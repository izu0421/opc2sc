from pycirclize import Circos
from typing import Optional

def make_chord_plot(
    df: pd.DataFrame,
    top_interacting: Optional[int] = 15,
    top_classifications: Optional[int] = 15,
    out_file: str = "chord_plot.png",
    dpi: int = 300,
    label_size: int = 15,       # bigger labels
):
    # Drop missing
    df = df.dropna(subset=["classification", "Interacting_cell"]).copy()
    df["classification"] = df["classification"].astype(str).str.strip()
    df["Interacting_cell"] = df["Interacting_cell"].astype(str).str.strip()
    

    # Count pair occurrences
    pair_counts = (
        df.value_counts(["Interacting_cell", "classification"])
          .rename("n")
          .reset_index()
    )

    # ⛔ exclude weak links (n < 2)
    # pair_counts = pair_counts[pair_counts["n"] >= 2]

    # Optionally limit rows
    if top_interacting is not None:
        keep_rows = (
            pair_counts.groupby("Interacting_cell")["n"]
            .sum()
            .sort_values(ascending=False)
            .head(top_interacting).index
        )
        pair_counts = pair_counts[pair_counts["Interacting_cell"].isin(keep_rows)]

    # Optionally limit columns
    if top_classifications is not None:
        keep_cols = (
            pair_counts.groupby("classification")["n"]
            .sum()
            .sort_values(ascending=False)
            .head(top_classifications).index
        )
        pair_counts = pair_counts[pair_counts["classification"].isin(keep_cols)]

    # Order rows/columns by frequency
    row_order = (
        pair_counts.groupby("Interacting_cell")["n"].sum()
        .sort_values(ascending=False).index
    )
    col_order = (
        pair_counts.groupby("classification")["n"].sum()
        .sort_values(ascending=True).index
    )

    # Build matrix
    matrix_df = (
        pair_counts
        .pivot(index="Interacting_cell", columns="classification", values="n")
        .fillna(0)
        .astype(int)
        .reindex(index=row_order, columns=col_order)
    )

    # Plot chord diagram
    circos = Circos.chord_diagram(
        matrix_df,
        space=4,
        cmap="tab20",
        label_kws=dict(size=label_size),
        link_kws=dict(ec="black", lw=0.5, alpha=0.9),
    )

    circos.savefig(out_file, dpi=dpi)
    print(f"Chord diagram saved → {out_file}")
