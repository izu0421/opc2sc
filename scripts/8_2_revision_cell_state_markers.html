def plot_top_markers(adata, group="celltype", n_genes=5, save_loc=None):
    """
    Find marker genes for each group and plot top markers as dotplot.
    
    Parameters:
    -----------
    adata : AnnData
        Annotated data object
    group : str
        Column name in adata.obs to group by (default: "celltype")
    n_genes : int
        Number of top marker genes to plot (default: 5)
    save_loc : str, optional
        Path to save the figure (e.g., "figures/invivo_markers_dotplot.pdf")
    """
    # Compute marker genes
    sc.tl.rank_genes_groups(adata, groupby=group, method='wilcoxon')
    
    # Plot dotplot of top marker genes
    dp = sc.pl.dotplot(adata, 
                       var_names=sc.get.rank_genes_groups_df(adata, group=None).groupby('group').head(n_genes).groupby('group')['names'].apply(list).to_dict(),
                       groupby=group,
                       cmap='RdPu',
                       standard_scale='var',
                       show=False)
    
    if save_loc:
        plt.savefig(save_loc, bbox_inches='tight')
        plt.show()
        plt.close()
    else:
        plt.show()
    
    return adata
